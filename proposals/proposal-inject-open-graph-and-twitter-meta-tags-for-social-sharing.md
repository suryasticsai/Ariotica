# Inject Open Graph and Twitter meta tags for social sharing

Adds OG and Twitter meta tags to the generated index.html so links shared on Twitter, Discord and other platforms show rich previews with a dynamic image generated via Pollinations AI.

## Proposed Code
```javascript
module.exports = {\n  name: 'inject-open-graph',\n  execute: async (context) => {\n    let html = context.files['index.html'];\n    if (!html) return;\n    const titleMatch = html.match(/<title[^>]*>([^<]*)<\/title>/i);\n    const title = titleMatch ? titleMatch[1].trim() : 'Ariotic App';\n    const descMatch = html.match(/<meta[^>]+name=[\"']description[\"'][^>]*content=[\"']([^\"']*)[\"']/i);\n    const description = descMatch ? descMatch[1].trim() : `${title} - generated by Ariotica`;\n    const prompt = encodeURIComponent(`Open Graph image for ${title}`);\n    const ogImage = `https://image.pollinations.ai/prompt/${prompt}`;\n    const metaTags = `\n    <meta property=\"og:title\" content=\"${title}\">\n    <meta property=\"og:description\" content=\"${description}\">\n    <meta property=\"og:image\" content=\"${ogImage}\">\n    <meta property=\"og:type\" content=\"website\">\n    <meta name=\"twitter:card\" content=\"summary_large_image\">\n    <meta name=\"twitter:title\" content=\"${title}\">\n    <meta name=\"twitter:description\" content=\"${description}\">\n    <meta name=\"twitter:image\" content=\"${ogImage}\">`;;\n    if (/<\/head>/i.test(html)) {\n      if (!/property=[\"']og:title[\"']/i.test(html)) {\n        html = html.replace(/<\/head>/i, metaTags + '\n</head>');\n      }\n    } else {\n      html = metaTags + '\n' + html;\n    }\n    context.files['index.html'] = html;\n  }\n};
```
